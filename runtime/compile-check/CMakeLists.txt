cmake_minimum_required(VERSION 3.16)
project(xran_sdk_check LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 14)
find_package(Protobuf REQUIRED)
find_package(gRPC CONFIG REQUIRED)
find_package(cpprestsdk CONFIG REQUIRED)
find_package(etcd-cpp-api CONFIG REQUIRED)
find_package(spdlog CONFIG REQUIRED)
find_package(nlohmann_json CONFIG REQUIRED)
find_package(opentelemetry-cpp CONFIG REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(NETWORK REQUIRED IMPORTED_TARGET libdpdk rohc libwebsockets libprotobuf-c)
protobuf_generate_cpp(PROTO_SRCS PROTO_HDRS check.proto)
add_executable(check main.cc ${PROTO_SRCS})
target_include_directories(check PRIVATE ${CMAKE_CURRENT_BINARY_DIR})
target_link_libraries(check PRIVATE protobuf::libprotobuf gRPC::grpc++ cpprestsdk::cpprest
    etcd-cpp-api spdlog::spdlog nlohmann_json::nlohmann_json PkgConfig::NETWORK
    opentelemetry-cpp::trace opentelemetry-cpp::otlp_grpc_exporter opentelemetry-cpp::otlp_http_exporter
    ${CMAKE_DL_LIBS})
enable_testing()
add_test(NAME sdk-link-and-run COMMAND check)
add_executable(protobuf-static-check static.cc ${PROTO_SRCS})
target_include_directories(protobuf-static-check PRIVATE ${CMAKE_CURRENT_BINARY_DIR} ${Protobuf_INCLUDE_DIRS})
target_link_libraries(protobuf-static-check PRIVATE /usr/local/protobuf/lib/libprotobuf.a Threads::Threads ZLIB::ZLIB)
add_test(NAME protobuf-static-link-and-run COMMAND protobuf-static-check)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_executable(dpdk-lcores-check dpdk-lcores.c)
target_link_libraries(dpdk-lcores-check PRIVATE PkgConfig::NETWORK)
foreach(lcore 144 255 256)
    add_test(NAME dpdk-lcore-${lcore} COMMAND ${Python3_EXECUTABLE}
        ${CMAKE_CURRENT_SOURCE_DIR}/check-dpdk-lcores.py $<TARGET_FILE:dpdk-lcores-check> ${lcore})
    set_tests_properties(dpdk-lcore-${lcore} PROPERTIES TIMEOUT 45)
endforeach()
